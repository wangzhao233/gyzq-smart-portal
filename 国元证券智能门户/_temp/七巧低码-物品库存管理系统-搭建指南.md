## 四、关键脚本

### 4.1 出入库提交后更新库存（4分）

```javascript
(function() {
    var doc = $.context.getCurrentDocument();
    var appId = $.context.getCurrentApplicationId();
    
    var itemId = doc.getElementByName("物品").getValue();
    var operationType = doc.getElementByName("操作类型").getValue();
    var quantity = parseFloat(doc.getElementByName("数量").getValue());
    
    if (itemId == null || operationType == null || isNaN(quantity)) return;
    
    // 查询物品表当前库存
    var filterMap = new Packages.java.util.HashMap();
    filterMap.put("_id", itemId);
    var itemDocs = $.form.getFormDocumentsByFieldNameAndValue(appId, "物品信息", filterMap);
    
    if (itemDocs != null && itemDocs.size() > 0) {
        var itemDoc = itemDocs.get(0);
        var currentStock = parseFloat(itemDoc.getElementByName("当前库存").getValue()) || 0;
        var initStock = parseFloat(itemDoc.getElementByName("初始库存").getValue()) || 0;
        
        if (operationType.indexOf("入库") >= 0) {
            currentStock += quantity;
        } else {
            currentStock -= quantity;
            if (currentStock < 0) {
                $.message.sendGeneralStationMessage("库存不足", "操作失败：库存不足，当前库存为" + (currentStock + quantity), $.context.getCurrentUserId());
                return;
            }
        }
        
        // 更新物品库存
        itemDoc.getElementByName("当前库存").setValue(currentStock);
        $.form.saveFormDocument(itemDoc, appId);
        
        // 检查安全库存预警
        var safetyStock = parseFloat(itemDoc.getElementByName("安全库存").getValue()) || 0;
        if (safetyStock > 0 && currentStock <= safetyStock) {
            $.message.sendGeneralStationMessage(
                "库存预警",
                "物品【" + itemDoc.getElementByName("物品名称").getValue() + "】当前库存" + currentStock + "，已低于安全库存" + safetyStock,
                $.context.getCurrentUserId()
            );
        }
    }
})();
```

### 指标配置（2分）
| 指标 | 计算方式 | 来源 |
|------|---------|------|
| 总入库 | SUM(数量) WHERE 操作类型=入库 | 出入库记录 |
| 总出库 | SUM(数量) WHERE 操作类型=出库 | 出入库记录 |
| 总使用 | SUM(使用数量) | 使用记录 |

## 八、校验规则

```javascript
// 出入库提交前校验：库存不能为负
(function() {
    var doc = $.context.getCurrentDocument();
    var type = doc.getElementByName("操作类型").getValue();
    var qty = parseFloat(doc.getElementByName("数量").getValue()) || 0;
    
    if (type.indexOf("出库") >= 0) {
        var itemId = doc.getElementByName("物品").getValue();
        var appId = $.context.getCurrentApplicationId();
        var fm = new Packages.java.util.HashMap();
        fm.put("_id", itemId);
        var items = $.form.getFormDocumentsByFieldNameAndValue(appId, "物品信息", fm);
        if (items != null && items.size() > 0) {
            var stock = parseFloat(items.get(0).getElementByName("当前库存").getValue()) || 0;
            if (stock < qty) {
                $.message.sendGeneralStationMessage("校验失败", "库存不足，当前库存" + stock + "，出库数量" + qty, $.context.getCurrentUserId());
                return false;
            }
        }
    }
    return true;
})();
```

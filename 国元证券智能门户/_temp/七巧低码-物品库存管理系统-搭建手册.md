## 第3步：配置聚合表（6分）

进入"表单设计"→选择"物品信息"→编辑"当前库存"字段→改为聚合字段

### 聚合配置

| 配置项 | 值 |
|--------|-----|
| **数据源（1分）** | 出入库记录表（关联字段=物品） |
| **聚合指标（2分）** | 入库总量 = SUM(数量) WHERE 操作类型=入库 |
| | 出库总量 = SUM(数量) WHERE 操作类型=出库 |
| | 使用总量 = SUM(使用数量) WHERE 使用记录.物品=当前物品 |
| | 当前库存 = 初始库存 + 入库总量 - 出库总量 - 使用总量 |
| **查询字段（2分）** | 关联显示：物品名称、分类名称、规格型号 |

### 聚合表校验（1分）

在聚合字段公式中加校验：
```javascript
if (当前库存 < 0) return "库存异常";
return 当前库存;
```

## 第4步：配置脚本

### 脚本1：出入库提交后更新库存（4分）

**位置**：表单设计 → 出入库记录 → 脚本 → 提交执行后

```javascript
(function() {
    var doc = $.context.getCurrentDocument();
    var appId = $.context.getCurrentApplicationId();
    
    // 读取表单字段
    var itemId = doc.getElementByName("物品").getValue();
    var operationType = doc.getElementByName("操作类型").getValue();
    var quantity = parseFloat(doc.getElementByName("数量").getValue()) || 0;
    
    if (!itemId || !operationType || quantity <= 0) return;
    
    // 1. 查询物品当前库存（更新表单，1分）
    var filterMap = new Packages.java.util.HashMap();
    filterMap.put("_id", itemId);
    var items = $.form.getFormDocumentsByFieldNameAndValue(appId, "物品信息", filterMap);
    
    if (!items || items.size() === 0) {
        $.log.warn("未找到物品: " + itemId);
        return;
    }
    
    var item = items.get(0);
    var currentStock = parseFloat(item.getElementByName("当前库存").getValue()) || 0;
    
    // 2. 更新库存（更新信息，2分）
    if (operationType.indexOf("入库") >= 0) {
        currentStock += quantity;
    } else {
        // 筛选条件：检查库存是否充足（1分）
        if (currentStock < quantity) {
            $.message.sendGeneralStationMessage(
                "操作失败",
                "库存不足！当前库存：" + currentStock + "，出库数量：" + quantity,
                $.context.getCurrentUserId()
            );
            return;
        }
        currentStock -= quantity;
    }
    
    // 更新物品表库存字段
    item.getElementByName("当前库存").setValue(currentStock);
    $.form.saveFormDocument(item, appId);
    
    // 3. 安全库存预警
    var safetyStock = parseFloat(item.getElementByName("安全库存").getValue()) || 0;
    if (safetyStock > 0 && currentStock <= safetyStock) {
        $.message.sendGeneralStationMessage(
            "库存预警 ⚠️",
            "物品【" + item.getElementByName("物品名称").getValue() + "】当前库存 " + currentStock + "，已低于安全库存 " + safetyStock,
            $.context.getCurrentUserId()
        );
    }
})();
```

### 脚本2：使用记录提交后更新库存

**位置**：表单设计 → 使用记录 → 脚本 → 提交执行后

```javascript
(function() {
    var doc = $.context.getCurrentDocument();
    var appId = $.context.getCurrentApplicationId();
    
    var itemId = doc.getElementByName("物品").getValue();
    var useQuantity = parseFloat(doc.getElementByName("使用数量").getValue()) || 0;
    
    if (!itemId || useQuantity <= 0) return;
    
    var filterMap = new Packages.java.util.HashMap();
    filterMap.put("_id", itemId);
    var items = $.form.getFormDocumentsByFieldNameAndValue(appId, "物品信息", filterMap);
    
    if (!items || items.size() === 0) return;
    
    var item = items.get(0);
    var stock = parseFloat(item.getElementByName("当前库存").getValue()) || 0;
    
    if (stock < useQuantity) {
        $.message.sendGeneralStationMessage(
            "操作失败",
            "库存不足！当前库存：" + stock + "，使用数量：" + useQuantity,
            $.context.getCurrentUserId()
        );
        return;
    }
    
    item.getElementByName("当前库存").setValue(stock - useQuantity);
    $.form.saveFormDocument(item, appId);
})();
```

### 脚本3：提交前校验（聚合表校验1分）

**位置**：出入库记录表单 → 脚本 → 提交执行前

```javascript
(function() {
    var doc = $.context.getCurrentDocument();
    var type = doc.getElementByName("操作类型").getValue();
    var qty = parseFloat(doc.getElementByName("数量").getValue()) || 0;
    
    if (qty <= 0) {
        $.message.sendGeneralStationMessage("校验失败", "数量必须大于0", $.context.getCurrentUserId());
        return false;
    }
    
    if (type && type.indexOf("出库") >= 0) {
        var itemId = doc.getElementByName("物品").getValue();
        var appId = $.context.getCurrentApplicationId();
        var fm = new Packages.java.util.HashMap();
        fm.put("_id", itemId);
        var items = $.form.getFormDocumentsByFieldNameAndValue(appId, "物品信息", fm);
        if (items && items.size() > 0) {
            var stock = parseFloat(items.get(0).getElementByName("当前库存").getValue()) || 0;
            if (stock < qty) {
                $.message.sendGeneralStationMessage(
                    "校验失败",
                    "库存不足，当前库存" + stock + "，出库数量" + qty,
                    $.context.getCurrentUserId()
                );
                return false;
            }
        }
    }
    return true;
})();
```
